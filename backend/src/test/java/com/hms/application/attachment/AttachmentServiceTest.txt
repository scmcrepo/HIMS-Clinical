    @BeforeEach

        attachmentId = UUID.randomUUID();

        tenantId = UUID.randomUUID();

        attachment.setId(attachmentId);
        attachment.setFileName("test.jpg");

        attachment.setContentType("image/jpeg");
        attachment.setCategory("GENERAL");

        mockedFiles = mockStatic(Files.class);
        mockedTenantContext = mockStatic(TenantContext.class);

    @AfterEach
    void tearDown() {
        if (mockedFiles != null && !mockedFiles.isClosed()) {
            mockedFiles.close();
        }
        if (mockedTenantContext != null && !mockedTenantContext.isClosed()) {
            mockedTenantContext.close();
        }

    @Test
    void saveAttachment_ShouldSaveAndReturnNewAttachment_WhenTypeIsConsultantAndNoExisting() throws IOException {
        setupMockFile();

        Attachment result = attachmentService.saveAttachment(multipartFile, AttachmentType.CONSULTANT, encounterId, patientId, providerId, "GENERAL");

        assertNotNull(result);
        verify(attachmentRepo).save(any(Attachment.class));
    }

    @Test

    @Test
    void saveAttachment_ShouldSaveAndReturnNewAttachment_WhenCategoryIsHospitalLogoAndTenantExists() throws IOException {
        setupMockFile();
        mockedTenantContext.when(TenantContext::get).thenReturn(tenantId);
        
        Attachment oldLogo = new Attachment();
        oldLogo.setFilePath("/var/hms/attachments/old.jpg");
        when(attachmentRepo.findAllByCategoryAndTenantNative("HOSPITAL_LOGO", tenantId)).thenReturn(List.of(oldLogo));
        mockedFiles.when(() -> Files.deleteIfExists(any(Path.class))).thenReturn(true);
        when(attachmentRepo.save(any(Attachment.class))).thenReturn(attachment);

        Attachment result = attachmentService.saveAttachment(multipartFile, AttachmentType.VISIT, encounterId, patientId, providerId, "HOSPITAL_LOGO");

        assertNotNull(result);
        verify(attachmentRepo).delete(oldLogo);
        verify(attachmentRepo).flush();
        verify(attachmentRepo).save(any(Attachment.class));
    }
    
    @Test
    void saveAttachment_ShouldSaveAndReturnNewAttachment_WhenCategoryIsHospitalLogoAndTenantExists_DeleteFails() throws IOException {
        setupMockFile();

        assertNotNull(result);

        verify(attachmentRepo).flush();
        verify(attachmentRepo).save(any(Attachment.class));
    }

    @Test
    void downloadFile_ShouldThrowException_WhenAttachmentNotFound() {
        when(attachmentRepo.findById(attachmentId)).thenReturn(Optional.empty());

        assertThrows(ResourceNotFoundException.class, () -> attachmentService.downloadFile(attachmentId));
    }

    @Test
    void downloadFile_ShouldReturnResource_WhenFileExistsAndReadable() throws Exception {
        // Need to use a real path that exists (like temp directory) to avoid mocking UrlResource/File systems deeply
        Path tempFile = Files.createTempFile("test-hms-attachment", ".jpg");
        attachment.setFilePath(tempFile.toAbsolutePath().toString());
        
        when(attachmentRepo.findById(attachmentId)).thenReturn(Optional.of(attachment));
        // Pause mock for Files so UrlResource can read the real file
        mockedFiles.close();
        
        Resource result = attachmentService.downloadFile(attachmentId);

        assertNotNull(result);
        assertTrue(result.exists());
        assertTrue(result.isReadable());
        
        Files.deleteIfExists(tempFile);
        mockedFiles = mockStatic(Files.class); // restart mock for teardown
    }
    
    @Test
    void downloadFile_ShouldThrowException_WhenFileDoesNotExist() {
        attachment.setFilePath("/invalid/path/that/does/not/exist/12345.jpg");
        when(attachmentRepo.findById(attachmentId)).thenReturn(Optional.of(attachment));
        
        mockedFiles.close();
        assertThrows(BusinessRuleViolationException.class, () -> attachmentService.downloadFile(attachmentId));
        mockedFiles = mockStatic(Files.class);
    }

    @Test
    void deleteAttachment_ShouldDeleteAttachmentAndFile() {
        when(attachmentRepo.findById(attachmentId)).thenReturn(Optional.of(attachment));
        mockedFiles.when(() -> Files.deleteIfExists(any(Path.class))).thenReturn(true);

        attachmentService.deleteAttachment(attachmentId);

        mockedFiles.verify(() -> Files.deleteIfExists(any(Path.class)));
        verify(attachmentRepo).delete(attachment);
    }
    
    @Test
    void deleteAttachment_ShouldDeleteAttachment_EvenIfFileDeleteThrowsIOException() {
        when(attachmentRepo.findById(attachmentId)).thenReturn(Optional.of(attachment));
        mockedFiles.when(() -> Files.deleteIfExists(any(Path.class))).thenThrow(new IOException("File locked"));

        attachmentService.deleteAttachment(attachmentId);

        mockedFiles.verify(() -> Files.deleteIfExists(any(Path.class)));
        verify(attachmentRepo).delete(attachment);
    }

    // -------------------------------------------------------------------------
    // Helper

    private void setupMockFile() throws IOException {
        when(multipartFile.isEmpty()).thenReturn(false);
        when(multipartFile.getOriginalFilename()).thenReturn("test@#$file.jpg");
        when(multipartFile.getContentType()).thenReturn("image/jpeg");
        
        InputStream is = new ByteArrayInputStream("test data".getBytes());
        when(multipartFile.getInputStream()).thenReturn(is);

        mockedFiles.when(() -> Files.createDirectories(any(Path.class))).thenReturn(Paths.get("/var/hms/attachments"));
        mockedFiles.when(() -> Files.copy(any(InputStream.class), any(Path.class), any(StandardCopyOption.class))).thenReturn(100L);
    }
