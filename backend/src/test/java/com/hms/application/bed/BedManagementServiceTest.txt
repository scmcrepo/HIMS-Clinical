    @Test
    void createBed_ShouldCreateBed_WhenValidRequest() {
        CreateBedRequest request = new CreateBedRequest("Bed 1", roomCategoryId, com.hms.domain.shared.model.EntityStatus.ACTIVE);
        mockedTenantContext.when(TenantContext::get).thenReturn(tenantId);
        mockedBranchContext.when(BranchContext::get).thenReturn(branchId);
        
        when(bedRepo.findByTenantIdAndBranchIdAndNameIgnoreCase(tenantId, branchId, "Bed 1")).thenReturn(Optional.empty());
        when(bedRepo.save(any(Bed.class))).thenReturn(bed);
        when(roomCategoryRepo.findById(roomCategoryId)).thenReturn(Optional.of(new RoomCategory()));

        BedResponse response = bedManagementService.createBed(request);

        assertNotNull(response);
        assertEquals("Bed 1", response.name());
        verify(bedRepo).save(any(Bed.class));
    }

    @Test
    void createBed_ShouldThrowException_WhenBedNameExists() {

    @Test
    void allocateBed_ShouldAllocateBed_WhenValidRequest() {
        AllocateBedRequest request = new AllocateBedRequest(bedId, encounterId, java.util.UUID.randomUUID(), java.util.UUID.randomUUID(), LocalDate.now(), "CASH", java.util.UUID.randomUUID());
        
        when(bedRepo.findActiveByIdForUpdate(bedId)).thenReturn(Optional.of(bed));
        when(encounterRepo.findById(encounterId)).thenReturn(Optional.of(encounter));
        when(occupancyRepo.findActiveByEncounterId(encounterId)).thenReturn(Optional.empty());
        when(bedRepo.save(any(Bed.class))).thenReturn(bed);
        when(encounterRepo.save(any(ClinicalEncounter.class))).thenReturn(encounter);
        when(occupancyRepo.save(any(BedOccupancy.class))).thenReturn(occupancy);
        
        BedOccupancyResponse expectedResp = new BedOccupancyResponse(occupancy.getId(), bedId, encounterId, java.util.UUID.randomUUID(), Instant.now(), Instant.now(), true);
        when(bedMapper.toOccupancyResponse(any(BedOccupancy.class))).thenReturn(expectedResp);

        mockedTxManager.when(TransactionSynchronizationManager::isActualTransactionActive).thenReturn(false);

        BedOccupancyResponse response = bedManagementService.allocateBed(request);

        assertNotNull(response);
        verify(bedRepo).save(bed);
        verify(encounterRepo).save(encounter);
        verify(occupancyRepo).save(any(BedOccupancy.class));
        verify(bedBillingHelper).autoInjectBedCharge(any(), any(), any(), any(), any(), any(), any());
    }

    @Test
    void allocateBed_ShouldThrowException_WhenAdmissionDateInFuture() {

    // -------------------------------------------------------------------------
    // transferBed
    // -------------------------------------------------------------------------

    @Test
    void transferBed_ShouldTransferSuccessfully() {
        UUID newBedId = UUID.randomUUID();
        Bed newBed = new Bed();
        newBed.setId(newBedId);
        newBed.setBedStatus(BedStatus.AVAILABLE);

        when(bedRepo.findActiveByIdForUpdate(newBedId)).thenReturn(Optional.of(newBed));
        when(encounterRepo.findById(encounterId)).thenReturn(Optional.of(encounter));
        when(occupancyRepo.findActiveByEncounterId(encounterId)).thenReturn(Optional.of(occupancy));
        when(bedRepo.findActiveByIdForUpdate(bedId)).thenReturn(Optional.of(bed));
        when(occupancyRepo.save(any(BedOccupancy.class))).thenReturn(new BedOccupancy());
        
        BedOccupancyResponse expectedResp = new BedOccupancyResponse(occupancy.getId(), newBedId, encounterId, java.util.UUID.randomUUID(), Instant.now(), Instant.now(), true);
        when(bedMapper.toOccupancyResponse(any(BedOccupancy.class))).thenReturn(expectedResp);
        
        mockedTxManager.when(TransactionSynchronizationManager::isActualTransactionActive).thenReturn(false);

        BedOccupancyResponse response = bedManagementService.transferBed(encounterId, newBedId, LocalDate.now());

        assertNotNull(response);
        verify(occupancyRepo).save(occupancy); // old occupancy closed
        verify(bedRepo).save(bed); // old bed released
        verify(bedRepo).save(newBed); // new bed allocated
        verify(encounterRepo).saveAndFlush(encounter);
    }

        verify(encounterRepo, times(2)).save(encounter); // once in releaseBed, once in vacateBed
